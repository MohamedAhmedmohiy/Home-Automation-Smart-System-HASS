Paradigmatic Shift in Local Agency Note

Unlike cloud-reliant switches that fail during internet outages, HASS runs its logic on the ESP32 controller. Switching uses zero-crossing solid-state relays and needs no external API.

03Kernel Logic & Zero-Crossing Dispatch

Inspect the high-speed load shed routine executed directly on the ESP32 micro-kernel or the higher-level Java state manager:

// Interrupt Service Routine bound to AC Phase Detect (GPIO 14)
void IRAM_ATTR onZeroCrossInterrupt() {
    uint32_t current_micros = esp_timer_get_time();
    if (g_load_shed_requested && !g_triac_gate_tripped) {
        // Fire triac at exactly 0V sine point to extinguish arc
        gpio_set_level(PIN_TRIAC_GATE_CH1, 0);
        g_triac_gate_tripped = true;
 
        // Broadcast deterministic actuation via local telemetry bus
        xQueueSendFromISR(h_bus_events, &current_micros, NULL);
    }
}
 
// Diurnal evaluation loop executing every 1,000ms
void evaluateDiurnalThreshold(LoadChannel* ch) {
    if (ch->instant_current_ma < ch->vampire_threshold && ch->state == RELAY_ENABLED) {
        if (ch->idle_counter++ > IDLE_GRACE_PERIOD_CYCLES) {
            ESP_LOGW("HASS_POWER", "Vampire current detected: %d mA. Triggering shed", ch->instant_current_ma);
            g_load_shed_requested = true;
        }
    }
}
// HassDaemon.java - Higher-level state manager
public class HassDaemon implements BusListener {
    private final LoadRegistry registry;
    public void onZeroCrossEvent(long micros) {
        // Sync cloud shadow state after hardware actuation
        registry.getActiveChannels().forEach(ch -> {
            Firebase.setBool(fbdo, "/devices/relay/" + ch.id, ch.state == RELAY_OFF);
        });
    }
}
